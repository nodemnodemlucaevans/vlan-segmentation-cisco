! ==========================================
! CONFIGURATION SWITCH ACCES - NODEM LUCA EVANS
! Keyce Informatique & IA - Project VLAN
! ==========================================

enable
configure terminal

! 1. Création des VLANs
vlan 10
 name ADMINISTRATION
vlan 20
 name EMPLOYES
vlan 30
 name GUESTS
vlan 99
 name MANAGEMENT
exit

! 2. Configuration du Trunk vers le Routeur
interface GigabitEthernet0/1
 description Interconnexion Routeur Inter-VLAN
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30,99
 switchport nonegotiate
exit

! 3. Configuration des ports d'accès (Employés)
interface range FastEthernet0/11 - 20
 description Ports Utilisateurs Employes
 switchport mode access
 switchport access vlan 20
 switchport port-security
 switchport port-security maximum 2
 switchport port-security violation restrict
 switchport port-security mac-address sticky
 spanning-tree portfast
exit

! 4. Sécurisation des ports inutilisés
interface range FastEthernet0/21 - 24
 shutdown
 description Ports inactifs - Securise
exit

end
write memory
